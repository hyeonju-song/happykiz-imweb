(function () {
  var root = document.getElementById('hkSub');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  /* ── 주소 #resp 등으로 들어오면 해당 클리닉 강조 ── */
  function hit() {
    var id = (location.hash || '').replace('#', '');
    Array.prototype.forEach.call(root.querySelectorAll('.hk_cp'), function (c) { c.classList.remove('hk_hit'); });
    if (!id) return;
    var el = document.getElementById(id);
    if (!el || !root.contains(el) || !el.classList.contains('hk_cp')) return;
    el.classList.add('hk_rv', 'hk_in');
    el.classList.add('hk_hit');
    setTimeout(function () { el.scrollIntoView({ behavior: 'smooth', block: 'center' }); }, 250);
    setTimeout(function () { el.classList.remove('hk_hit'); }, 3200);
  }
  window.addEventListener('hashchange', hit);

  /* ── 스크롤 등장 ── */
  var els = root.querySelectorAll('.hk_rv');
  if (!('IntersectionObserver' in window)) { Array.prototype.forEach.call(els, function (e) { e.classList.add('hk_in'); }); hit(); return; }
  var io = new IntersectionObserver(function (entries) {
    entries.forEach(function (en) { if (en.isIntersecting) { en.target.classList.add('hk_in'); io.unobserve(en.target); } });
  }, { threshold: 0.08 });
  Array.prototype.forEach.call(els, function (e) { io.observe(e); });
  hit();
})();
