<script>
(function () {
  var root = document.getElementById('hkSub');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  /* ── 자가 체크리스트 ── */
  var msg = root.querySelector('.hk_ckmsg');
  var totalEl = root.querySelector('.hk_total');
  function update() {
    var total = 0;
    Array.prototype.forEach.call(root.querySelectorAll('.hk_ck'), function (ck) {
      var btns = ck.querySelectorAll('.hk_ckl button');
      var n = 0;
      Array.prototype.forEach.call(btns, function (b) { if (b.getAttribute('aria-pressed') === 'true') n++; });
      ck.querySelector('.hk_cnt').textContent = n + ' / ' + btns.length;
      ck.classList[n ? 'add' : 'remove']('hk_has');
      total += n;
    });
    if (totalEl) totalEl.textContent = total + '개';
    if (msg) msg.classList[total ? 'add' : 'remove']('hk_on');
  }
  Array.prototype.forEach.call(root.querySelectorAll('.hk_ckl button'), function (b) {
    b.addEventListener('click', function () {
      b.setAttribute('aria-pressed', b.getAttribute('aria-pressed') === 'true' ? 'false' : 'true');
      update();
    });
  });

  /* ── 스크롤 등장 ── */
  var els = root.querySelectorAll('.hk_rv');
  if (!('IntersectionObserver' in window)) { Array.prototype.forEach.call(els, function (e) { e.classList.add('hk_in'); }); return; }
  var io = new IntersectionObserver(function (entries) {
    entries.forEach(function (en) { if (en.isIntersecting) { en.target.classList.add('hk_in'); io.unobserve(en.target); } });
  }, { threshold: 0.08 });
  Array.prototype.forEach.call(els, function (e) { io.observe(e); });
})();
</script>
