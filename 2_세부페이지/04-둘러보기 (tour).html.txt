  figs.forEach(function (f) {
    f.setAttribute('tabindex', '0');
    f.setAttribute('role', 'button');
    f.style.cursor = 'zoom-in';
    f.addEventListener('click', function () { open(shown.indexOf(f)); });
    f.addEventListener('keydown', function (e) {
      if (e.key === 'Enter' || e.key === ' ') { e.preventDefault(); open(shown.indexOf(f)); }
    });
  });

  root.querySelector('#hkLbPrev').addEventListener('click', function (e) { e.stopPropagation(); show(idx - 1); });
  root.querySelector('#hkLbNext').addEventListener('click', function (e) { e.stopPropagation(); show(idx + 1); });
  root.querySelector('#hkLbClose').addEventListener('click', close);
  lb.addEventListener('click', function (e) { if (e.target === lb || e.target.className === 'hk_stage') close(); });

  document.addEventListener('keydown', function (e) {
    if (!lb.classList.contains('hk_open')) return;
    if (e.key === 'Escape') close();
    else if (e.key === 'ArrowLeft') show(idx - 1);
    else if (e.key === 'ArrowRight') show(idx + 1);
  });

  /* 모바일 좌우 밀기 */
  var sx = 0, sy = 0;
  lb.addEventListener('touchstart', function (e) { sx = e.touches[0].clientX; sy = e.touches[0].clientY; }, { passive: true });
  lb.addEventListener('touchend', function (e) {
    var dx = e.changedTouches[0].clientX - sx, dy = e.changedTouches[0].clientY - sy;
    if (Math.abs(dx) > 50 && Math.abs(dx) > Math.abs(dy)) show(idx + (dx < 0 ? 1 : -1));
  }, { passive: true });
})();
</script>
