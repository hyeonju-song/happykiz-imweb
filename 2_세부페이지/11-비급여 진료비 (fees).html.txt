(function () {
  var root = document.getElementById('hkSub');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  /* ── 스크롤 등장 ── */
  var els = root.querySelectorAll('.hk_rv');
  if (!('IntersectionObserver' in window)) { Array.prototype.forEach.call(els, function (e) { e.classList.add('hk_in'); }); }
  else {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (en) { if (en.isIntersecting) { en.target.classList.add('hk_in'); io.unobserve(en.target); } });
    }, { threshold: 0.12 });
    Array.prototype.forEach.call(els, function (e) { io.observe(e); });
  }

  /* ── 비급여 표 그리기 ── */
  var tabsEl = root.querySelector('.hk_ftabs');
  var listEl = root.querySelector('.hk_flist');
  var countEl = root.querySelector('.hk_fcount');
  var noneEl = root.querySelector('.hk_fnone');
  var input = root.querySelector('.hk_fsearch input');
  var clr = root.querySelector('.hk_fclr');
  if (!tabsEl || !listEl || !window.HK_FEES) return;

  function esc(s) { return String(s).replace(/[&<>"]/g, function (c) { return { '&':'&amp;', '<':'&lt;', '>':'&gt;', '"':'&quot;' }[c]; }); }
  function price(p) {
    p = String(p).trim();
    if (p === '무료') return '<span class="hk_fp hk_free">무료</span>';
    if (p.indexOf('~') > -1) return '<span class="hk_fp">' + esc(p.replace(/\s*~\s*/, ' ~ ')) + '<small>원</small></span>';
    return '<span class="hk_fp">' + esc(p) + '<small>원</small></span>';
  }

  var tabHtml = '', listHtml = '', total = 0;
  HK_FEES.forEach(function (t, ti) {
    var n = 0;
    t.groups.forEach(function (g) {
      n += g.items.length;
      listHtml += '<div class="hk_fg" data-tab="' + t.key + '">' +
        '<div class="hk_fgh"><b>' + esc(g.name) + ' <i>' + g.items.length + '개</i></b>' +
        '<span class="hk_ftag hk_hide">' + esc(t.name) + '</span>' +
        (g.date ? '<small>최종수정일 ' + esc(g.date) + '</small>' : '') + '</div>';
      g.items.forEach(function (it) {
        listHtml += '<div class="hk_fr" data-q="' + esc((it[0] + ' ' + g.name + ' ' + t.name).toLowerCase().replace(/\s+/g, '')) + '">' +
          '<span class="hk_fn">' + esc(it[0]) + '</span>' +
          '<span class="hk_fu">' + esc(it[1] || '') + '</span>' + price(it[2]) + '</div>';
      });
      listHtml += '</div>';
    });
    total += n;
    tabHtml += '<button type="button" class="hk_ftab" role="tab" data-tab="' + t.key + '">' + esc(t.name) + ' <i>' + n + '</i></button>';
  });
  tabsEl.innerHTML = tabHtml;
  listEl.innerHTML = listHtml;

  var tabs = tabsEl.querySelectorAll('.hk_ftab');
  var groups = listEl.querySelectorAll('.hk_fg');
  var names = listEl.querySelectorAll('.hk_fn');
  names = Array.prototype.map.call(names, function (el) { return { el: el, txt: el.textContent }; });
  var cur = HK_FEES[0].key;

  function showTab(key) {
    cur = key;
    Array.prototype.forEach.call(tabs, function (b) {
      var on = b.getAttribute('data-tab') === key;
      b.classList[on ? 'add' : 'remove']('hk_on');
      b.setAttribute('aria-selected', on ? 'true' : 'false');
    });
    render();
  }

  function render() {
    var q = (input.value || '').toLowerCase().replace(/\s+/g, '');
    var hit = 0;
    tabsEl.classList[q ? 'add' : 'remove']('hk_dim');
    clr.classList[input.value ? 'add' : 'remove']('hk_on');
    Array.prototype.forEach.call(groups, function (g) {
      var inTab = g.getAttribute('data-tab') === cur;
      var tag = g.querySelector('.hk_ftag');
      if (tag) tag.classList[q ? 'remove' : 'add']('hk_hide');
      var any = 0;
      Array.prototype.forEach.call(g.querySelectorAll('.hk_fr'), function (r) {
        var show = q ? r.getAttribute('data-q').indexOf(q) > -1 : inTab;
        r.classList[show ? 'remove' : 'add']('hk_hide');
        if (show) any++;
      });
      g.classList[any ? 'remove' : 'add']('hk_hide');
      hit += any;
    });
    /* 검색어 강조 */
    var raw = (input.value || '').trim();
    names.forEach(function (o) {
      if (!raw) { o.el.textContent = o.txt; return; }
      var i = o.txt.toLowerCase().indexOf(raw.toLowerCase());
      o.el.innerHTML = i < 0 ? esc(o.txt) : esc(o.txt.slice(0, i)) + '<mark>' + esc(o.txt.slice(i, i + raw.length)) + '</mark>' + esc(o.txt.slice(i + raw.length));
    });
    countEl.innerHTML = q ? '검색 결과 <b>' + hit + '</b>건 (전체 ' + total + '개 항목 중)' : '<b>' + hit + '</b>개 항목';
    noneEl.classList[hit === 0 ? 'add' : 'remove']('hk_on');
  }

  Array.prototype.forEach.call(tabs, function (b) {
    b.addEventListener('click', function () {
      if (input.value) input.value = '';
      showTab(b.getAttribute('data-tab'));
    });
  });
  input.addEventListener('input', render);
  clr.addEventListener('click', function () { input.value = ''; render(); input.focus(); });

  /* 주소 뒤 #cert 등으로 탭 바로 열기 */
  function fromHash() {
    var h = (location.hash || '').replace('#', '');
    for (var i = 0; i < HK_FEES.length; i++) if (HK_FEES[i].key === h) {
      showTab(h);
      setTimeout(function () { tabsEl.scrollIntoView({ behavior: 'smooth', block: 'start' }); }, 300);
      return true;
    }
    return false;
  }
  if (!fromHash()) showTab(cur);
  window.addEventListener('hashchange', fromHash);
})();
</script>
