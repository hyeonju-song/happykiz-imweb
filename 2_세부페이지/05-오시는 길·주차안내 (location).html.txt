<script>
(function () {
  var root = document.getElementById('hkSub');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  /* ── 주소 복사 ── */
  var btn = root.querySelector('.hk_copy');
  if (btn) {
    btn.addEventListener('click', function () {
      var addr = btn.getAttribute('data-hk-addr') || '';
      var txt = btn.querySelector('.hk_copytxt');

      function done() {
        btn.classList.add('hk_done');
        if (txt) txt.textContent = '복사되었습니다';
        setTimeout(function () {
          btn.classList.remove('hk_done');
          if (txt) txt.textContent = '주소 복사';
        }, 1800);
      }

      if (navigator.clipboard && navigator.clipboard.writeText) {
        navigator.clipboard.writeText(addr).then(done, fallback);
      } else {
        fallback();
      }

      /* 구형 브라우저 · http 환경 대비 */
      function fallback() {
        try {
          var ta = document.createElement('textarea');
          ta.value = addr;
          ta.setAttribute('readonly', '');
          ta.style.position = 'fixed';
          ta.style.left = '-9999px';
          document.body.appendChild(ta);
          ta.select();
          document.execCommand('copy');
          document.body.removeChild(ta);
          done();
        } catch (e) {}
      }
    });
  }

  /* ── 스크롤 등장 ── */
  var els = root.querySelectorAll('.hk_rv');
  if (!('IntersectionObserver' in window)) { Array.prototype.forEach.call(els, function (e) { e.classList.add('hk_in'); }); return; }
  var io = new IntersectionObserver(function (entries) {
    entries.forEach(function (en) { if (en.isIntersecting) { en.target.classList.add('hk_in'); io.unobserve(en.target); } });
  }, { threshold: 0.12 });
  Array.prototype.forEach.call(els, function (e) { io.observe(e); });
})();
</script>
