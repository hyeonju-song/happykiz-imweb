<script>
(function () {
  var root = document.getElementById('hkSub');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';
  var els = root.querySelectorAll('.hk_rv');
  if (!('IntersectionObserver' in window)) { Array.prototype.forEach.call(els, function (e) { e.classList.add('hk_in'); }); return; }
  var io = new IntersectionObserver(function (entries) {
    entries.forEach(function (en) { if (en.isIntersecting) { en.target.classList.add('hk_in'); io.unobserve(en.target); } });
  }, { threshold: 0.08 });
  Array.prototype.forEach.call(els, function (e) { io.observe(e); });
})();
</script>
