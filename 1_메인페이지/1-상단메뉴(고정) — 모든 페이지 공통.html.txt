(function () {
  var root = document.getElementById('hkHd');
  if (!root || root.dataset.hkReady) return;   /* 아임웹에서 두 번 실행되는 것 방지 */
  root.dataset.hkReady = '1';

  /* 스크롤하면 헤더를 조금 낮게 */
  function onScroll() {
    if (window.pageYOffset > 40) root.classList.add('hk_solid');
    else root.classList.remove('hk_solid');
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  /* PC 전체메뉴 — 마우스를 올리면 세부 페이지가 펼쳐집니다 */
  var gnbItems = root.querySelectorAll('.hk_gnb > li');
  var cols = root.querySelectorAll('.hk_mcol');
  var timer = null;

  function hot(g) {
    for (var i = 0; i < gnbItems.length; i++) gnbItems[i].classList[gnbItems[i].getAttribute('data-g') === g ? 'add' : 'remove']('hk_hot');
    for (var j = 0; j < cols.length; j++) cols[j].classList[cols[j].getAttribute('data-g') === g ? 'add' : 'remove']('hk_hot');
  }
  function open(g) { if (window.innerWidth <= 1024) return; clearTimeout(timer); root.classList.add('hk_open'); hot(g); }
  function close(delay) {
    clearTimeout(timer);
    timer = setTimeout(function () { root.classList.remove('hk_open'); hot(null); }, delay === undefined ? 140 : delay);
  }
  Array.prototype.forEach.call(gnbItems, function (li) {
    var g = li.getAttribute('data-g');
    li.addEventListener('mouseenter', function () { open(g); });
    var a = li.querySelector('a');
    if (a) a.addEventListener('focus', function () { open(g); });
  });
  Array.prototype.forEach.call(cols, function (c) { c.addEventListener('mouseenter', function () { open(c.getAttribute('data-g')); }); });
  root.addEventListener('mouseenter', function () { clearTimeout(timer); });
  root.addEventListener('mouseleave', function () { close(); });
  document.addEventListener('keydown', function (e) { if (e.key === 'Escape') close(0); });

  /* 모바일 메뉴 + 아코디언 */
  var burger = root.querySelector('.hk_burger');
  var groups = root.querySelectorAll('.hk_mg');
  burger.addEventListener('click', function () {
    var on = root.classList.toggle('hk_mo');
    burger.setAttribute('aria-expanded', on ? 'true' : 'false');
    burger.setAttribute('aria-label', on ? '메뉴 닫기' : '메뉴 열기');
    document.documentElement.style.overflow = on ? 'hidden' : '';
    if (on && groups.length && !root.querySelector('.hk_mg.hk_on')) groups[0].classList.add('hk_on');
  });
  Array.prototype.forEach.call(groups, function (g) {
    g.querySelector('.hk_mh').addEventListener('click', function () {
      var was = g.classList.contains('hk_on');
      Array.prototype.forEach.call(groups, function (x) { x.classList.remove('hk_on'); });
      if (!was) g.classList.add('hk_on');
    });
  });
  Array.prototype.forEach.call(root.querySelectorAll('.hk_mm a'), function (a) {
    a.addEventListener('click', function () {
      root.classList.remove('hk_mo');
      document.documentElement.style.overflow = '';
      burger.setAttribute('aria-expanded', 'false');
    });
  });
})();
