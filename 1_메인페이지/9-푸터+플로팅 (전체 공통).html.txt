<script>
(function () {
  var root = document.getElementById('hkQm');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  var panel  = root.querySelector('.hk_panel');
  var toggle = root.querySelector('.hk_toggle');
  var top    = root.querySelector('.hk_top');

  function setOpen(on) {
    panel.classList[on ? 'add' : 'remove']('hk_on');
    toggle.setAttribute('aria-expanded', on ? 'true' : 'false');
  }
  /* 홈 첫 화면(메인 슬라이드)에서는 펼침. 아래로 내려가면 본문을 가리지 않게 접음
     세부페이지는 상단 카드를 가리므로 본문 옆에 자리가 있는 넓은 화면에서만 펼침
     (직접 누른 뒤에는 자동으로 바꾸지 않음) */
  var wide = window.matchMedia('(min-width:1760px)');
  var home = document.querySelector('#hkHero .hk_stage');
  var touched = false, atTop = !!home;
  function auto() { if (!touched) setOpen(atTop || wide.matches); }
  if (wide.addEventListener) wide.addEventListener('change', auto); else wide.addListener(auto);
  auto();

  toggle.addEventListener('click', function () {
    touched = true;
    setOpen(!panel.classList.contains('hk_on'));
  });

  top.addEventListener('click', function () {
    try { window.scrollTo({ top: 0, behavior: 'smooth' }); }
    catch (e) { window.scrollTo(0, 0); }
  });

  function onScroll() {
    if (window.pageYOffset > 250) top.classList.add('hk_show');
    else top.classList.remove('hk_show');
  }
  window.addEventListener('scroll', onScroll, { passive: true });
  onScroll();

  /* 푸터가 보이면 퀵메뉴를 감춰 푸터 메뉴를 가리지 않게 합니다 */
  var ft = document.getElementById('hkFt');
  if (ft && window.IntersectionObserver) {
    new IntersectionObserver(function (es) {
      es.forEach(function (e) { root.classList[e.isIntersecting ? 'add' : 'remove']('hk_away'); });
    }, { threshold: 0 }).observe(ft);
  }

  /* 홈 슬라이드가 보이는지 확인 */
  if (home && window.IntersectionObserver) {
    new IntersectionObserver(function (es) {
      es.forEach(function (e) { atTop = e.isIntersecting; auto(); });
    }, { threshold: 0, rootMargin: '-120px 0px 0px 0px' }).observe(home);
  }
})();
</script>
