(function () {
  var root = document.getElementById('hkNt');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  var tabs = root.querySelectorAll('.hk_tab'), panels = root.querySelectorAll('.hk_panel');
  Array.prototype.forEach.call(tabs, function (t) {
    t.addEventListener('click', function () {
      var k = t.getAttribute('data-t');
      Array.prototype.forEach.call(tabs, function (x) { x.classList[x === t ? 'add' : 'remove']('hk_on'); });
      Array.prototype.forEach.call(panels, function (p) { p.classList[p.getAttribute('data-p') === k ? 'add' : 'remove']('hk_on'); });
    });
  });

  var els = root.querySelectorAll('.hk_rv');
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches || !window.IntersectionObserver) {
    Array.prototype.forEach.call(els, function (el) { el.classList.add('hk_in'); }); return;
  }
  var obs = new IntersectionObserver(function (es) {
    es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('hk_in'); obs.unobserve(e.target); } });
  }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
  Array.prototype.forEach.call(els, function (el) { obs.observe(el); });
})();
</script>
