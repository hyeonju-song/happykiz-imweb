(function () {
  var root = document.getElementById('hkLoc');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  /* ── 카카오맵(다음 약도) ──
     roughmapLoader.js 는 document.write 로 Lander 를 불러오는데,
     아임웹 코드 위젯은 페이지 로드 후 삽입되어 document.write 가 무시됩니다.
     그래서 로더가 넣어 준 설정값(cdn/phase)으로 Lander 를 직접 붙입니다. */
  (function () {
    var box = root.querySelector('#daumRoughmapContainer1607930741646');
    if (!box) return;

    function add(src, cls) {
      if (document.querySelector('script[data-hk-km="' + cls + '"]')) return;
      var s = document.createElement('script');
      s.charset = 'UTF-8';
      s.src = src;
      s.setAttribute('data-hk-km', cls);
      if (cls === 'loader') s.className = 'daum_roughmap_loader_script';
      document.head.appendChild(s);
    }

    function render() {
      if (box.children.length) return;
      try {
        new daum.roughmap.Lander({
          'timestamp': '1607930741646',
          'key': '23gkj',
          'mapWidth': '100%',
          'mapHeight': '400'
        }).render();
      } catch (e) {}
    }

    var n = 0;
    function tick() {
      var rm = window.daum && window.daum.roughmap;
      if (rm && rm.Lander) { render(); return; }
      if (rm && rm.cdn && rm.phase) {
        add((rm.url_protocal || 'https:') + '//t1.kakaocdn.net/kakaomapweb/roughmap/place/'
            + rm.phase + '/' + rm.cdn + '/roughmapLander.js', 'lander');
      }
      if (++n > 120) return;
      setTimeout(tick, 150);
    }

    /* 18초가 지나도 지도가 안 붙으면 지도 칸을 치우고 '카카오맵에서 보기' 안내만 남김
       (그 전에도 지도가 비어 있는 동안은 아래에 깔린 안내가 그대로 보입니다)
       지도는 iframe 이 아니라 .wrap_map 안에 직접 그려지므로 그것으로 확인합니다. */
    setTimeout(function () {
      if (!box.querySelector('.wrap_map')) box.parentNode.classList.add('hk_nomap');
    }, 18000);

    add('https://ssl.daumcdn.net/dmaps/map_js_init/roughmapLoader.js', 'loader');
    tick();
  })();

  var els = root.querySelectorAll('.hk_rv');
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches || !window.IntersectionObserver) {
    Array.prototype.forEach.call(els, function (el) { el.classList.add('hk_in'); }); return;
  }
  var obs = new IntersectionObserver(function (es) {
    es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('hk_in'); obs.unobserve(e.target); } });
  }, { threshold: 0.1, rootMargin: '0px 0px -40px 0px' });
  Array.prototype.forEach.call(els, function (el) { obs.observe(el); });
})();
