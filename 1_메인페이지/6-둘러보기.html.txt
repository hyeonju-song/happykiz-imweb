<script>
(function () {
  var root = document.getElementById('hkEnv');
  if (!root || root.dataset.hkReady) return;
  root.dataset.hkReady = '1';

  var els = root.querySelectorAll('.hk_rv');
  if (window.matchMedia('(prefers-reduced-motion: reduce)').matches || !window.IntersectionObserver) {
    Array.prototype.forEach.call(els, function (el) { el.classList.add('hk_in'); });
  } else {
    var obs = new IntersectionObserver(function (es) {
      es.forEach(function (e) { if (e.isIntersecting) { e.target.classList.add('hk_in'); obs.unobserve(e.target); } });
    }, { threshold: 0.12, rootMargin: '0px 0px -40px 0px' });
    Array.prototype.forEach.call(els, function (el) { obs.observe(el); });
  }

  var rail = root.querySelector('.hk_rail');
  var fill = root.querySelector('.hk_fill');
  var prev = root.querySelector('.hk_prev');
  var next = root.querySelector('.hk_next');
  function paint() {
    var max = rail.scrollWidth - rail.clientWidth;
    var w = Math.max((rail.clientWidth / rail.scrollWidth) * 100, 12);
    var p = max > 0 ? rail.scrollLeft / max : 0;
    fill.style.width = w + '%';
    fill.style.left = (p * (100 - w)) + '%';
    if (prev && next) { prev.disabled = rail.scrollLeft <= 2; next.disabled = rail.scrollLeft >= max - 2; }
  }
  rail.addEventListener('scroll', paint, { passive: true });
  window.addEventListener('resize', paint);
  function stepSize() {
    var card = rail.querySelector('.hk_card');
    if (!card) return rail.clientWidth * 0.8;
    return card.getBoundingClientRect().width + 18;
  }
  /* 브라우저 기본 부드러운 스크롤이 막혀 있는 환경이 있어 직접 애니메이션합니다 */
  var animId = null, animTimer = null, target = null;
  function finish() {
    if (animId) { cancelAnimationFrame(animId); animId = null; }
    if (animTimer) { clearTimeout(animTimer); animTimer = null; }
    if (target !== null) { rail.scrollLeft = target; target = null; }
    rail.classList.remove('hk_anim');
    paint();
  }
  function animateTo(to) {
    var max = rail.scrollWidth - rail.clientWidth;
    if (to < 0) to = 0;
    if (to > max) to = max;
    if (animId) { cancelAnimationFrame(animId); animId = null; }
    if (animTimer) { clearTimeout(animTimer); animTimer = null; }
    var start = rail.scrollLeft, diff = to - start, t0 = null, dur = 420;
    if (Math.abs(diff) < 1) { finish(); return; }
    target = to;
    rail.classList.add('hk_anim');
    animTimer = setTimeout(finish, dur + 260);
    function frame(ts) {
      if (t0 === null) t0 = ts;
      var p = (ts - t0) / dur; if (p > 1) p = 1;
      var e = 1 - Math.pow(1 - p, 3);
      rail.scrollLeft = start + diff * e;
      if (p < 1) animId = requestAnimationFrame(frame); else { animId = null; finish(); }
    }
    animId = requestAnimationFrame(frame);
  }
  function go(dir) {
    var base = (target === null) ? rail.scrollLeft : target;
    if (window.requestAnimationFrame) animateTo(base + stepSize() * dir); else rail.scrollLeft = base + stepSize() * dir;
  }
  if (prev) prev.addEventListener('click', function () { go(-1); });
  if (next) next.addEventListener('click', function () { go(1); });

  /* 마우스로 잡아끌어서 넘기기 */
  var down = false, startX = 0, startLeft = 0, moved = false;
  rail.addEventListener('pointerdown', function (e) {
    if (e.pointerType !== 'mouse' || e.button !== 0) return;
    if (animId || animTimer) finish();
    down = true; moved = false; startX = e.clientX; startLeft = rail.scrollLeft;
    rail.classList.add('hk_dragging');
    try { rail.setPointerCapture(e.pointerId); } catch (err) {}
  });
  rail.addEventListener('pointermove', function (e) {
    if (!down) return;
    var dx = e.clientX - startX;
    if (Math.abs(dx) > 4) moved = true;
    rail.scrollLeft = startLeft - dx; e.preventDefault();
  });
  function endDrag(e) {
    if (!down) return;
    down = false; rail.classList.remove('hk_dragging');
    try { rail.releasePointerCapture(e.pointerId); } catch (err) {}
    paint();
  }
  rail.addEventListener('pointerup', endDrag);
  rail.addEventListener('pointercancel', endDrag);
  rail.addEventListener('lostpointercapture', endDrag);
  rail.addEventListener('click', function (e) { if (moved) { e.preventDefault(); e.stopPropagation(); moved = false; } }, true);
  rail.addEventListener('dragstart', function (e) { e.preventDefault(); });

  paint();
  window.addEventListener('load', paint);
})();
</script>
